interface ArtifactVersion {

  timestamp: number;

const versionStore = new Map<string, ArtifactVersion[]>();

function pushVersion(artifactId: string, code: string) {
  const existing = versionStore.get(artifactId) || [];
  if (existing.length > 0 && existing[existing.length - 1].code === code) return;
  versionStore.set(artifactId, [...existing, { code, timestamp: Date.now() }]);
}

function getVersions(artifactId: string): ArtifactVersion[] {
  return versionStore.get(artifactId) || [];
}

function escapeHtml(str: string): string {
  return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
}

function fileExtensionFor(language: ArtifactLanguage): string {

function prismLanguageFor(language: ArtifactLanguage): string {

let prismLoadPromise: Promise<void> | null = null;

function loadPrism(): Promise<void> {
  if (typeof window === "undefined") return Promise.resolve();
  if ((window as any).Prism) return Promise.resolve();
  if (prismLoadPromise) return prismLoadPromise;

  prismLoadPromise = new Promise((resolve) => {
    const cssId = "prism-theme-css";
    if (!document.getElementById(cssId)) {
      const link = document.createElement("link");
      link.id = cssId;
      link.rel = "stylesheet";
      link.href = "[https://cdnjs.cloudflare.com/ajax/libs/prism/1.29.0/themes/prism-tomorrow.min.css](https://cdnjs.cloudflare.com/ajax/libs/prism/1.29.0/themes/prism-tomorrow.min.css)";
      document.head.appendChild(link);
    }

    const coreScript = document.createElement("script");
    coreScript.src = "[https://cdnjs.cloudflare.com/ajax/libs/prism/1.29.0/components/prism-core.min.js](https://cdnjs.cloudflare.com/ajax/libs/prism/1.29.0/components/prism-core.min.js)";
    coreScript.onload = () => {
      const autoloader = document.createElement("script");
      autoloader.src =
        "[https://cdnjs.cloudflare.com/ajax/libs/prism/1.29.0/plugins/autoloader/prism-autoloader.min.js](https://cdnjs.cloudflare.com/ajax/libs/prism/1.29.0/plugins/autoloader/prism-autoloader.min.js)";
      autoloader.onload = () => resolve();
      autoloader.onerror = () => resolve();
      document.body.appendChild(autoloader);
    };
    coreScript.onerror = () => resolve();
    document.body.appendChild(coreScript);
  });

  return prismLoadPromise;
}

const DEFAULT_SAMPLE_CODE = \`<!DOCTYPE html>

export function ArtifactCard({
  title = "مثال Artifact",
  language = "html",
  code = DEFAULT_SAMPLE_CODE,
  artifactId,
}: Partial<ArtifactCardProps>) {

interface ArtifactViewerProps extends ArtifactCardProps {
  onClose: () => void;
}

function ArtifactViewer({ title, language, code, artifactId, onClose }: ArtifactViewerProps) {
  const [view, setView] = useState<"preview" | "code">("preview");
  const [isFullscreen, setIsFullscreen] = useState(false);
  const [reloadKey, setReloadKey] = useState(0);
  const [isLoading, setIsLoading] = useState(true);
  const [copied, setCopied] = useState(false);
  const [prismReady, setPrismReady] = useState(false);
  const codeRef = useRef<HTMLElement>(null);

  const versions = useMemo(
    () => (artifactId ? getVersions(artifactId) : [{ code, timestamp: Date.now() }]),
    [artifactId, code]
  );
  const [versionIndex, setVersionIndex] = useState(versions.length - 1);
  const activeCode = versions[versionIndex]?.code ?? code;
  const srcDoc = useMemo(() => buildIframeSrcDoc(language, activeCode), [language, activeCode]);

  useEffect(() => {
    if (view === "code") {
      loadPrism().then(() => setPrismReady(true));
    }
  }, [view]);

  useEffect(() => {
    if (prismReady && codeRef.current && (window as any).Prism) {
      (window as any).Prism.highlightElement(codeRef.current);
    }
  }, [prismReady, activeCode, view]);

  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      if (e.key === "Escape") onClose();
    };
    window.addEventListener("keydown", handler);
    return () => window.removeEventListener("keydown", handler);
  }, [onClose]);

  const handleReload = useCallback(() => {
    setIsLoading(true);
    setReloadKey((k) => k + 1);
  }, []);

  const handleCopy = useCallback(async () => {
    try {
      await navigator.clipboard.writeText(activeCode);
      setCopied(true);
      setTimeout(() => setCopied(false), 1500);
    } catch {
      /* ignore */
    }
  }, [activeCode]);

  const handleDownload = useCallback(() => {
    const ext = fileExtensionFor(language);
    const blob = new Blob([activeCode], { type: "text/plain" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = \`\${(title || "artifact").replace(/\\s+/g, "-").toLowerCase()}.\${ext}\`;
    document.body.appendChild(a);
    a.click();
    a.remove();
    URL.revokeObjectURL(url);
  }, [activeCode, language, title]);

  const containerStyle: React.CSSProperties = isFullscreen
    ? { position: "fixed", inset: 0, zIndex: 1000, background: THEME.bg, display: "flex", flexDirection: "column" }
    : {
        position: "fixed",
        inset: 0,
        zIndex: 1000,
        background: "rgba(10,8,20,0.65)",
        display: "flex",
        alignItems: "center",
        justifyContent: "center",
      };

  const panelStyle: React.CSSProperties = isFullscreen
    ? { width: "100%", height: "100%", display: "flex", flexDirection: "column", background: THEME.bg }
    : {
        width: "92%",
        maxWidth: 920,
        height: "82%",
        background: THEME.bg,
        borderRadius: 16,
        display: "flex",
        flexDirection: "column",
        overflow: "hidden",
        boxShadow: "0 25px 70px rgba(0,0,0,0.5)",
        border: \`1px solid \${THEME.border}\`,
      };

  return (
    <div style={containerStyle} onClick={!isFullscreen ? onClose : undefined}>
      <div style={panelStyle} onClick={(e) => e.stopPropagation()}>
        <div
          style={{
            display: "flex",
            alignItems: "center",
            justifyContent: "space-between",
            padding: "12px 18px",
            borderBottom: \`1px solid \${THEME.border}\`,
            flexWrap: "wrap",
            gap: 8,
            background: THEME.panelBg,
          }}
        >
          <div style={{ display: "flex", alignItems: "center", gap: 8 }}>
            <Icon.Bolt />
            <span style={{ fontWeight: 700, fontSize: 14, color: THEME.text }}>{title}</span>
          </div>

          <div style={{ display: "flex", gap: 4, alignItems: "center" }}>
            {versions.length > 1 && (
              <div style={{ display: "flex", alignItems: "center", gap: 2, marginInlineEnd: 8 }}>
                <button
                  disabled={versionIndex === 0}
                  onClick={() => setVersionIndex((i) => Math.max(0, i - 1))}
                  style={iconBtn(versionIndex === 0)}
                  title="النسخة السابقة"
                >
                  <Icon.ChevronR />
                </button>
                <span style={{ fontSize: 12, color: THEME.textMuted }}>
                  {versionIndex + 1}/{versions.length}
                </span>
                <button
                  disabled={versionIndex === versions.length - 1}
                  onClick={() => setVersionIndex((i) => Math.min(versions.length - 1, i + 1))}
                  style={iconBtn(versionIndex === versions.length - 1)}
                  title="النسخة التالية"
                >
                  <Icon.ChevronL />
                </button>
              </div>
            )}

            <div style={{ display: "flex", background: THEME.bg, borderRadius: 10, padding: 3 }}>
              <button onClick={() => setView("preview")} style={tabBtn(view === "preview")}>
                معاينة
              </button>
              <button onClick={() => setView("code")} style={tabBtn(view === "code")}>
                الكود
              </button>
            </div>

            <button onClick={handleReload} title="إعادة تشغيل المعاينة" style={iconBtn(false)}>
              <Icon.Play />
            </button>
            <button onClick={handleCopy} title="نسخ الكود" style={iconBtn(false)}>
              {copied ? <Icon.Check /> : <Icon.Copy />}
            </button>
            <button onClick={handleDownload} title="تحميل الملف" style={iconBtn(false)}>
              <Icon.Download />
            </button>
            <button onClick={() => setIsFullscreen((f) => !f)} title="ملء الشاشة" style={iconBtn(false)}>
              <Icon.Expand />
            </button>
            <button onClick={onClose} title="إغلاق (Esc)" style={iconBtn(false)}>
              <Icon.Close />
            </button>
          </div>
        </div>

        <div style={{ flex: 1, overflow: "hidden", position: "relative", background: THEME.bg }}>
          {view === "preview" ? (
            <>
              {isLoading && (
                <div
                  style={{
                    position: "absolute",
                    inset: 0,
                    display: "flex",
                    alignItems: "center",
                    justifyContent: "center",
                    background: THEME.bg,
                  }}
                >
                  <span style={{ color: THEME.textMuted, fontSize: 14 }}>جارٍ تحميل المعاينة...</span>
                </div>
              )}
              <iframe
                key={reloadKey}
                srcDoc={srcDoc}
                sandbox="allow-scripts"
                onLoad={() => setIsLoading(false)}
                style={{ width: "100%", height: "100%", border: "none", background: "#fff" }}
                title={title}
              />
            </>
          ) : (
            <div style={{ height: "100%", overflow: "auto", padding: 16 }}>
              <pre style={{ margin: 0, direction: "ltr", textAlign: "left", fontSize: 13 }}>
                <code ref={codeRef} className={\`language-\${prismLanguageFor(language)}\`}>
                  {activeCode}
                </code>
              </pre>
            </div>
          )}
        </div>
      </div>
    </div>
  );
}

function iconBtn(disabled: boolean): React.CSSProperties {
  return {
    border: "none",
    background: "none",
    cursor: disabled ? "default" : "pointer",
    opacity: disabled ? 0.3 : 1,
    padding: 7,
    display: "flex",
    alignItems: "center",
    borderRadius: 8,
  };
}

function tabBtn(active: boolean): React.CSSProperties {
  return {
    padding: "5px 12px",
    borderRadius: 8,
    border: "none",
    background: active ? THEME.tabActiveBg : "transparent",
    color: active ? THEME.text : THEME.textMuted,
    fontSize: 13,
    fontWeight: active ? 600 : 400,
    cursor: "pointer",
  };
}

export default ArtifactCard;
